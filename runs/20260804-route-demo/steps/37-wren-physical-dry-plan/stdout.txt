WITH device_orders AS (SELECT "wren_src_device_orders".product_l1 AS product_l1, "wren_src_device_orders".year_month AS year_month, "wren_src_device_orders".order_amount AS order_amount, "wren_src_device_orders".profit_amount AS profit_amount, "wren_src_device_orders".org_code AS org_code, "wren_src_device_orders"."status" AS status FROM (SELECT device_orders.order_amount, device_orders.org_code, device_orders.product_l1, device_orders.profit_amount, device_orders."status", device_orders.year_month FROM (SELECT __source.order_amount AS order_amount, __source.org_code AS org_code, __source.product_l1 AS product_l1, __source.profit_amount AS profit_amount, __source."status" AS "status", __source.year_month AS year_month FROM demo."main".device_orders AS __source) AS device_orders) AS "wren_src_device_orders") SELECT product_l1, SUM(CASE WHEN status = 'confirmed' THEN order_amount ELSE 0 END) AS device_order_amount, SUM(CASE WHEN status = 'confirmed' THEN profit_amount ELSE 0 END) AS device_profit_amount FROM device_orders WHERE org_code = 'ORG_1001' AND year_month >= 202601 AND year_month <= 202606 GROUP BY 1
