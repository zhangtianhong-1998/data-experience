SELECT
  device_order__product_l1
  , SUM(device_order_amount) AS device_order_amount
  , SUM(device_profit_amount) AS device_profit_amount
FROM (
  SELECT
    org_code AS device_order__org_code
    , product_l1 AS device_order__product_l1
    , CASE WHEN status = 'confirmed' THEN order_amount ELSE 0 END AS device_order_amount
    , CASE WHEN status = 'confirmed' THEN profit_amount ELSE 0 END AS device_profit_amount
  FROM "metricflow-device"."main"."device_orders" device_orders_src_10000
  WHERE DATE_TRUNC('day', order_date) BETWEEN '2026-01-01' AND '2026-06-30'
) subq_5
WHERE device_order__org_code = 'ORG_1001'
GROUP BY
  device_order__product_l1
ORDER BY device_order__product_l1
